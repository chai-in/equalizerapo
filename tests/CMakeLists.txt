cmake_minimum_required(VERSION 3.20)
project(EqualizerAPORegressionTests LANGUAGES CXX)
enable_testing()
option(ENABLE_SANITIZERS "Run with address and undefined-behavior sanitizers" OFF)
add_executable(convolution_tests convolution_tests.cpp ../filters/ConvolutionFilter.cpp)
target_compile_features(convolution_tests PRIVATE cxx_std_14)
# Only platform/IO/FFT boundaries are doubled. The production filter is compiled unchanged.
target_include_directories(convolution_tests PRIVATE stubs ..)
if(MSVC)
  target_compile_options(convolution_tests PRIVATE /W4 /EHsc)
else()
  target_compile_options(convolution_tests PRIVATE -Wall -Wextra -Wno-unknown-pragmas)
  if(ENABLE_SANITIZERS)
    target_compile_options(convolution_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(convolution_tests PRIVATE -fsanitize=address,undefined)
  endif()
endif()
foreach(case IN ITEMS valid short_reads missing wrong_rate empty negative_frames zero_channels negative_channels oversized invalid_block zero_block eof read_error reinitialize reopen_failure zero_process)
  add_test(NAME convolution_${case} COMMAND convolution_tests ${case})
  set_tests_properties(convolution_${case} PROPERTIES TIMEOUT 5)
  if(ENABLE_SANITIZERS AND NOT MSVC)
    set_tests_properties(convolution_${case} PROPERTIES ENVIRONMENT "ASAN_OPTIONS=alloc_dealloc_mismatch=1")
  endif()
endforeach()
